      case 'mining': return <PageMining/>;
      case 'reports': return <PageReports/>;
      case 'infra': return <PageInfra/>;
      case 'settings': return <PageSettings role={role} setRole={setRole}/>;

    }

    <div style={{display:'flex',minHeight:'100vh'}}>

    </div>
  );
};
